 /**
 * ============================================================
 * EQRMSS — Plan State Manager
 * ============================================================
 *
 * Purpose
 * -------
 * Authoritative state-transition service for the EQRMSS
 * Planner/Worker execution pipeline.
 *
 * Responsibilities
 * ----------------
 * - Read the canonical current-plan.json.
 * - Read the authoritative state-transition-rules.json.
 * - Validate legal plan-state transitions.
 * - Validate exact task identity.
 * - Validate persisted authorization scope.
 * - Persist authorized execution state.
 * - Persist execution/validation/completion state.
 * - Record audit metadata required by the transition contract.
 * - Reread persisted state after every mutation.
 * - Refuse unauthorized, ambiguous, substituted, or expanded work.
 * - Repair narrowly defined legacy authorization-state inconsistencies.
 *
 * Non-Responsibilities
 * --------------------
 * - Does not implement EQRMSS tasks.
 * - Does not edit EQRMSS source files.
 * - Does not invoke workers.
 * - Does not decide what a task should be.
 * - Does not infer task identity from repository contents.
 * - Does not infer authorization from conversation state.
 * - Does not modify frozen task definitions.
 * - Does not authorize a task as part of state repair.
 *
 * Architecture
 * ------------
 *
 * USER / LEAD
 *     |
 *     v
 * PLANNER
 *     |
 *     | request transition
 *     v
 * PLAN STATE MANAGER
 *     |
 *     +--> state-transition-rules.json
 *     |
 *     +--> current-plan.json
 *     |
 *     v
 * persisted authoritative state
 *
 * The Plan State Manager is the sole component responsible for
 * mutating current-plan.json.
 *
 * Critical Invariant
 * ------------------
 * A state transition is successful only when the resulting
 * current-plan.json has been persisted and reread successfully.
 *
 * A successful command therefore means:
 *
 *   validate -> mutate -> persist -> reread -> verify
 *
 * Never:
 *
 *   request -> assume success
 *
 * Recovery invariant
 * ------------------
 * Recovery is not authorization.
 *
 * A repair operation may only restore a known-safe canonical
 * state. It may not create authorization, expand scope, execute
 * work, or alter frozen task definitions.
 *
 * Usage
 * -----
 *
 * From the EQRMSS repository root:
 *
 *   node .agents/state/plan-state-manager.js status
 *
 *   node .agents/state/plan-state-manager.js authorize P001
 *
 *   node .agents/state/plan-state-manager.js beginExecution P001
 *
 *   node .agents/state/plan-state-manager.js beginValidation P001
 *
 *   node .agents/state/plan-state-manager.js completeTask P001
 *
 *   node .agents/state/plan-state-manager.js failValidation P001
 *
 *   node .agents/state/plan-state-manager.js revokeAuthorization
 *
 *   node .agents/state/plan-state-manager.js repairAuthorizationState
 *
 * All commands emit JSON to stdout.
 *
 * ============================================================
 */

"use strict";

const fs = require("node:fs");
const path = require("node:path");
const crypto = require("node:crypto");

const ROOT = path.resolve(__dirname, "../..");

const STATE_PATH = path.join(
  ROOT,
  ".agents",
  "state",
  "current-plan.json"
);

const RULES_PATH = path.join(
  ROOT,
  ".agents",
  "state",
  "state-transition-rules.json"
);

const COMMANDS = new Set([
  "status",
  "authorize",
  "beginExecution",
  "beginValidation",
  "completeTask",
  "failValidation",
  "revokeAuthorization",
  "repairAuthorizationState"
]);

/**
 * ------------------------------------------------------------
 * General utilities
 * ------------------------------------------------------------
 */

function now() {
  return new Date().toISOString();
}

function fail(code, message, details = {}) {
  const result = {
    ok: false,
    code,
    message,
    ...details
  };

  console.log(JSON.stringify(result, null, 2));
  process.exitCode = 1;
}

function succeed(result) {
  console.log(
    JSON.stringify(
      {
        ok: true,
        ...result
      },
      null,
      2
    )
  );
}

function deepClone(value) {
  return JSON.parse(JSON.stringify(value));
}

function stableJson(value) {
  return JSON.stringify(value);
}

function sha256(value) {
  return crypto
    .createHash("sha256")
    .update(value, "utf8")
    .digest("hex");
}

function readJson(filePath, label) {
  let raw;

  try {
    raw = fs.readFileSync(filePath, "utf8");
  } catch (error) {
    throw new StateManagerError(
      "CANNOT_READ_CANONICAL_STATE",
      `Unable to read ${label}.`,
      {
        path: path.relative(ROOT, filePath),
        cause: error.message
      }
    );
  }

  try {
    return JSON.parse(raw);
  } catch (error) {
    throw new StateManagerError(
      "INVALID_JSON",
      `${label} contains invalid JSON.`,
      {
        path: path.relative(ROOT, filePath),
        cause: error.message
      }
    );
  }
}

class StateManagerError extends Error {
  constructor(code, message, details = {}) {
    super(message);
    this.name = "StateManagerError";
    this.code = code;
    this.details = details;
  }
}

/**
 * ------------------------------------------------------------
 * State loading
 * ------------------------------------------------------------
 */

function loadState() {
  const state = readJson(
    STATE_PATH,
    "current-plan.json"
  );

  if (
    !state ||
    typeof state !== "object" ||
    Array.isArray(state)
  ) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json must contain a JSON object."
    );
  }

  if (!state.planId) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json is missing planId."
    );
  }

  if (!state.planVersion) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json is missing planVersion."
    );
  }

  if (!state.status) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json is missing plan status."
    );
  }

  if (
    !state.tasks ||
    typeof state.tasks !== "object"
  ) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json is missing the tasks object."
    );
  }

  if (
    !state.authorization ||
    typeof state.authorization !== "object"
  ) {
    throw new StateManagerError(
      "INVALID_STATE",
      "current-plan.json is missing the authorization object."
    );
  }

  return state;
}

function loadRules() {
  const rules = readJson(
    RULES_PATH,
    "state-transition-rules.json"
  );

  if (
    !rules ||
    typeof rules !== "object"
  ) {
    throw new StateManagerError(
      "INVALID_RULES",
      "state-transition-rules.json must contain a JSON object."
    );
  }

  if (
    !rules.transitions ||
    typeof rules.transitions !== "object"
  ) {
    throw new StateManagerError(
      "INVALID_RULES",
      "state-transition-rules.json is missing transitions."
    );
  }

  return rules;
}

/**
 * ------------------------------------------------------------
 * Rule validation
 * ------------------------------------------------------------
 */

function assertKnownPlanStatus(state, rules) {
  if (!rules.states?.[state.status]) {
    throw new StateManagerError(
      "UNKNOWN_PLAN_STATUS",
      `Unknown plan status: ${state.status}.`
    );
  }
}

function getTransition(
  rules,
  transitionName
) {
  const transition =
    rules.transitions?.[transitionName];

  if (!transition) {
    throw new StateManagerError(
      "UNKNOWN_TRANSITION",
      `Transition is not defined by the authoritative rules: ${transitionName}.`
    );
  }

  return transition;
}

function assertTransitionAllowed(
  state,
  rules,
  transitionName
) {
  const transition =
    getTransition(
      rules,
      transitionName
    );

  if (
    !transition.from.includes(
      state.status
    )
  ) {
    throw new StateManagerError(
      "INVALID_STATE_TRANSITION",
      `Transition '${transitionName}' is not permitted from state '${state.status}'.`,
      {
        transition: transitionName,
        currentStatus: state.status,
        allowedFrom: transition.from,
        targetStatus: transition.to
      }
    );
  }

  return transition;
}

/**
 * ------------------------------------------------------------
 * Canonical-state consistency
 * ------------------------------------------------------------
 *
 * The authoritative state machine distinguishes the plan's
 * top-level status from its authorization metadata.
 *
 * These combinations are intentionally rejected for normal
 * execution:
 *
 *   READY + AUTHORIZED
 *   AUTHORIZED + NOT_AUTHORIZED
 *
 * The first combination is the legacy/orphaned state that the
 * repair command is specifically designed to recover.
 * ------------------------------------------------------------
 */

function assertCanonicalStateConsistency(
  state,
  {
    allowLegacyAuthorizationState = false
  } = {}
) {
  const authorizationStatus =
    state.authorization?.status;

  const authorizationScope =
    state.authorization?.scope;

  if (
    state.status === "READY" &&
    authorizationStatus === "AUTHORIZED"
  ) {
    if (
      allowLegacyAuthorizationState
    ) {
      return;
    }

    throw new StateManagerError(
      "INCONSISTENT_CANONICAL_STATE",
      "Canonical plan state is READY while authorization is AUTHORIZED. The state requires controlled repair before normal transitions may continue.",
      {
        status: state.status,
        authorizationStatus,
        authorizationScope:
          Array.isArray(
            authorizationScope
          )
            ? authorizationScope
            : []
      }
    );
  }

  if (
    state.status === "AUTHORIZED" &&
    authorizationStatus !== "AUTHORIZED"
  ) {
    throw new StateManagerError(
      "INCONSISTENT_CANONICAL_STATE",
      "Canonical plan state is AUTHORIZED but authorization metadata is not AUTHORIZED.",
      {
        status: state.status,
        authorizationStatus
      }
    );
  }

  if (
    state.status === "READY" &&
    authorizationStatus === "NOT_AUTHORIZED" &&
    Array.isArray(authorizationScope) &&
    authorizationScope.length > 0
  ) {
    throw new StateManagerError(
      "INCONSISTENT_CANONICAL_STATE",
      "Canonical plan state is READY and NOT_AUTHORIZED but retains an authorization scope.",
      {
        status: state.status,
        authorizationStatus,
        authorizationScope
      }
    );
  }

  if (
    state.status === "AUTHORIZED" &&
    Array.isArray(authorizationScope) &&
    authorizationScope.length === 0
  ) {
    throw new StateManagerError(
      "INCONSISTENT_CANONICAL_STATE",
      "Canonical plan state is AUTHORIZED but contains an empty authorization scope."
    );
  }
}

/**
 * ------------------------------------------------------------
 * Authorization validation
 * ------------------------------------------------------------
 */

function ensureAuthorizationObject(state) {
  if (
    !state.authorization ||
    typeof state.authorization !== "object"
  ) {
    throw new StateManagerError(
      "MISSING_AUTHORIZATION",
      "Canonical plan does not contain a valid authorization object."
    );
  }

  if (
    !Array.isArray(
      state.authorization.scope
    )
  ) {
    throw new StateManagerError(
      "INVALID_AUTHORIZATION_SCOPE",
      "Authorization scope must be an array."
    );
  }
}

function getTask(state, taskId) {
  if (
    !taskId ||
    typeof taskId !== "string"
  ) {
    throw new StateManagerError(
      "MISSING_TASK_ID",
      "An exact task ID is required."
    );
  }

  const task = state.tasks?.[taskId];

  if (
    !task ||
    typeof task !== "object"
  ) {
    throw new StateManagerError(
      "UNKNOWN_TASK_ID",
      `Task '${taskId}' does not exist in the authoritative current plan.`,
      {
        taskId
      }
    );
  }

  return task;
}

function ensureExactTaskIdentity(
  state,
  taskId,
  handoff = {}
) {
  const task = getTask(
    state,
    taskId
  );

  const requiredFields = [
    "planId",
    "planVersion",
    "taskId",
    "taskTitle",
    "taskObjective",
    "worker"
  ];

  const expected = {
    planId: state.planId,
    planVersion: state.planVersion,
    taskId,
    taskTitle: task.title,
    taskObjective: task.objective,
    worker: task.worker
  };

  for (
    const field of requiredFields
  ) {
    if (
      handoff[field] === undefined
    ) {
      continue;
    }

    if (
      handoff[field] !==
      expected[field]
    ) {
      throw new StateManagerError(
        "TASK_IDENTITY_MISMATCH",
        `Worker handoff field '${field}' does not match authoritative plan state.`,
        {
          field,
          expected:
            expected[field],
          received:
            handoff[field]
        }
      );
    }
  }

  return task;
}

function assertTaskAuthorized(
  state,
  taskId
) {
  ensureAuthorizationObject(
    state
  );

  if (
    state.authorization.status !==
    "AUTHORIZED"
  ) {
    throw new StateManagerError(
      "MISSING_AUTHORIZATION",
      "The plan does not contain persisted AUTHORIZED state."
    );
  }

  if (
    !state.authorization.scope.includes(
      taskId
    )
  ) {
    throw new StateManagerError(
      "AUTHORIZATION_SCOPE_MISMATCH",
      `Task '${taskId}' is not present in persisted authorization scope.`,
      {
        taskId,
        authorizedScope:
          state.authorization.scope
      }
    );
  }
}

/**
 * ------------------------------------------------------------
 * Dependency validation
 * ------------------------------------------------------------
 */

function isTaskComplete(
  state,
  taskId
) {
  const task =
    state.tasks?.[taskId];

  if (
    !task ||
    typeof task !== "object"
  ) {
    return false;
  }

  return (
    task.status === "COMPLETE" ||
    task.executionStatus ===
      "COMPLETE" ||
    task.validationStatus ===
      "COMPLETE"
  );
}

function assertDependenciesSatisfied(
  state,
  task
) {
  const dependencies =
    Array.isArray(
      task.dependencies
    )
      ? task.dependencies
      : [];

  for (
    const dependency of dependencies
  ) {
    if (
      !isTaskComplete(
        state,
        dependency
      )
    ) {
      throw new StateManagerError(
        "DEPENDENCY_UNSATISFIED",
        `Task dependency '${dependency}' has not been established as complete.`,
        {
          dependency
        }
      );
    }
  }
}

/**
 * ------------------------------------------------------------
 * Frozen-plan integrity
 * ------------------------------------------------------------
 */

function buildPlanFingerprint(
  state
) {
  const material = {
    planId: state.planId,
    planVersion:
      state.planVersion,
    tasks: state.tasks
  };

  return sha256(
    stableJson(material)
  );
}

function ensureNoMaterialPlanChange(
  state
) {
  const recorded =
    state.authorization
      ?.planFingerprint ??
    state.execution
      ?.planFingerprint ??
    state.validation
      ?.planFingerprint;

  if (!recorded) {
    return;
  }

  const current =
    buildPlanFingerprint(
      state
    );

  if (
    recorded !== current
  ) {
    throw new StateManagerError(
      "MATERIAL_PLAN_CHANGE",
      "The frozen plan definition has changed since authorization/execution began.",
      {
        recordedFingerprint:
          recorded,
        currentFingerprint:
          current
      }
    );
  }
}

/**
 * ------------------------------------------------------------
 * Audit
 * ------------------------------------------------------------
 */

function ensureAuditContainers(
  state
) {
  if (
    !Array.isArray(
      state.auditHistory
    )
  ) {
    state.auditHistory = [];
  }
}

function appendAudit(
  state,
  {
    actor,
    action,
    previousState,
    newState,
    taskScope = []
  }
) {
  ensureAuditContainers(
    state
  );

  state.auditHistory.push({
    timestamp: now(),
    actor,
    action,
    planId:
      state.planId,
    planVersion:
      state.planVersion,
    previousState,
    newState,
    taskScope:
      Array.isArray(taskScope)
        ? [...taskScope]
        : []
  });
}

/**
 * ------------------------------------------------------------
 * Persistence
 * ------------------------------------------------------------
 */

function persistState(
  state
) {
  const directory =
    path.dirname(
      STATE_PATH
    );

  const temporaryPath =
    path.join(
      directory,
      `.current-plan.json.${process.pid}.${Date.now()}.tmp`
    );

  const contents =
    JSON.stringify(
      state,
      null,
      2
    ) + "\n";

  try {
    fs.writeFileSync(
      temporaryPath,
      contents,
      {
        encoding: "utf8",
        mode: 0o660
      }
    );

    fs.renameSync(
      temporaryPath,
      STATE_PATH
    );
  } catch (error) {
    try {
      if (
        fs.existsSync(
          temporaryPath
        )
      ) {
        fs.unlinkSync(
          temporaryPath
        );
      }
    } catch {
      // Preserve original persistence error.
    }

    throw new StateManagerError(
      "STATE_PERSISTENCE_FAILED",
      "Unable to persist current-plan.json.",
      {
        path:
          path.relative(
            ROOT,
            STATE_PATH
          ),
        cause:
          error.message
      }
    );
  }
}

function persistAndVerify(
  state,
  expected = {}
) {
  persistState(state);

  const reread =
    loadState();

  if (
    expected.status &&
    reread.status !==
      expected.status
  ) {
    throw new StateManagerError(
      "PERSISTENCE_VERIFICATION_FAILED",
      "Persisted state did not contain the expected plan status.",
      {
        expectedStatus:
          expected.status,
        actualStatus:
          reread.status
      }
    );
  }

  if (
    expected.authorizationStatus
  ) {
    if (
      reread.authorization
        ?.status !==
      expected.authorizationStatus
    ) {
      throw new StateManagerError(
        "PERSISTENCE_VERIFICATION_FAILED",
        "Persisted state did not contain the expected authorization status.",
        {
          expectedAuthorizationStatus:
            expected.authorizationStatus,
          actualAuthorizationStatus:
            reread
              .authorization
              ?.status
        }
      );
    }
  }

  if (
    expected.scope
  ) {
    const actual =
      reread.authorization
        ?.scope ?? [];

    if (
      stableJson(actual) !==
      stableJson(
        expected.scope
      )
    ) {
      throw new StateManagerError(
        "PERSISTENCE_VERIFICATION_FAILED",
        "Persisted authorization scope does not match the requested scope.",
        {
          expectedScope:
            expected.scope,
          actualScope:
            actual
        }
      );
    }
  }

  return reread;
}

/**
 * ------------------------------------------------------------
 * Transition: authorize
 * ------------------------------------------------------------
 */

function authorize(
  state,
  rules,
  taskId
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "authorize"
  );

  ensureAuthorizationObject(
    state
  );

  if (
    state.authorization.status !==
    "NOT_AUTHORIZED"
  ) {
    throw new StateManagerError(
      "ALREADY_AUTHORIZED",
      "The plan is already authorized."
    );
  }

  const task =
    getTask(
      state,
      taskId
    );

  const requestedScope = [
    taskId
  ];

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "AUTHORIZED";

  next.authorization = {
    ...next.authorization,
    status:
      "AUTHORIZED",
    scope:
      requestedScope,
    planFingerprint:
      buildPlanFingerprint(
        state
      ),
    authorizedAt:
      now()
  };

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "authorize",
      previousState,
      newState:
        "AUTHORIZED",
      taskScope:
        requestedScope
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "AUTHORIZED",
        authorizationStatus:
          "AUTHORIZED",
        scope:
          requestedScope
      }
    );

  return {
    transition:
      "authorize",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    taskId,
    taskTitle:
      task.title,
    authorizedScope:
      persisted.authorization
        .scope,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Transition: beginExecution
 * ------------------------------------------------------------
 */

function beginExecution(
  state,
  rules,
  taskId
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "beginExecution"
  );

  assertTaskAuthorized(
    state,
    taskId
  );

  ensureNoMaterialPlanChange(
    state
  );

  const task =
    ensureExactTaskIdentity(
      state,
      taskId
    );

  assertDependenciesSatisfied(
    state,
    task
  );

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "EXECUTING";

  next.execution = {
    ...(next.execution ?? {}),
    taskId,
    planId:
      next.planId,
    planVersion:
      next.planVersion,
    taskTitle:
      task.title,
    worker:
      task.worker,
    startedAt:
      now(),
    planFingerprint:
      next.authorization
        .planFingerprint ??
      buildPlanFingerprint(
        state
      )
  };

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "beginExecution",
      previousState,
      newState:
        "EXECUTING",
      taskScope:
        [taskId]
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "EXECUTING"
      }
    );

  return {
    transition:
      "beginExecution",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    taskId,
    worker:
      task.worker,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Transition: beginValidation
 * ------------------------------------------------------------
 */

function beginValidation(
  state,
  rules,
  taskId
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "beginValidation"
  );

  ensureNoMaterialPlanChange(
    state
  );

  const executingTaskId =
    state.execution?.taskId;

  if (!executingTaskId) {
    throw new StateManagerError(
      "MISSING_EXECUTING_TASK",
      "No executing task is recorded."
    );
  }

  if (
    executingTaskId !==
    taskId
  ) {
    throw new StateManagerError(
      "TASK_IDENTITY_MISMATCH",
      "Validation task ID does not match the executing task.",
      {
        executingTaskId,
        requestedTaskId:
          taskId
      }
    );
  }

  getTask(
    state,
    taskId
  );

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "VALIDATING";

  next.validation = {
    ...(next.validation ?? {}),
    taskId,
    planId:
      next.planId,
    planVersion:
      next.planVersion,
    startedAt:
      now(),
    planFingerprint:
      next.execution
        ?.planFingerprint ??
      buildPlanFingerprint(
        state
      )
  };

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "beginValidation",
      previousState,
      newState:
        "VALIDATING",
      taskScope:
        [taskId]
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "VALIDATING"
      }
    );

  return {
    transition:
      "beginValidation",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    taskId,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Transition: completeTask
 * ------------------------------------------------------------
 */

function completeTask(
  state,
  rules,
  taskId,
  evidence
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "completeTask"
  );

  ensureNoMaterialPlanChange(
    state
  );

  const executingTaskId =
    state.execution?.taskId;

  const validatingTaskId =
    state.validation?.taskId;

  if (
    executingTaskId !==
    taskId
  ) {
    throw new StateManagerError(
      "TASK_IDENTITY_MISMATCH",
      "Completion task ID does not match the executing task."
    );
  }

  if (
    validatingTaskId !==
    taskId
  ) {
    throw new StateManagerError(
      "TASK_IDENTITY_MISMATCH",
      "Completion task ID does not match the validating task."
    );
  }

  if (
    evidence?.validationPassed !==
    true
  ) {
    throw new StateManagerError(
      "VALIDATION_REQUIRED",
      "Task cannot be completed without explicit successful validation evidence."
    );
  }

  const task =
    getTask(
      state,
      taskId
    );

  const requiredChecks =
    Array.isArray(
      task.validation
    )
      ? task.validation
      : [];

  const suppliedChecks =
    Array.isArray(
      evidence.validationChecks
    )
      ? evidence.validationChecks
      : [];

  const suppliedById =
    new Map(
      suppliedChecks
        .filter(
          (check) =>
            check &&
            typeof check.id ===
              "string"
        )
        .map(
          (check) => [
            check.id,
            check
          ]
        )
    );

  for (
    const requiredCheck of
      requiredChecks
  ) {
    const checkId =
      typeof requiredCheck ===
      "string"
        ? requiredCheck
        : requiredCheck?.id;

    if (!checkId) {
      throw new StateManagerError(
        "INVALID_VALIDATION_EVIDENCE",
        "A required validation entry does not contain a usable ID."
      );
    }

    const supplied =
      suppliedById.get(
        checkId
      );

    if (
      !supplied ||
      supplied.passed !==
        true
    ) {
      throw new StateManagerError(
        "VALIDATION_CHECK_FAILED",
        `Required validation check '${checkId}' has not passed.`,
        {
          checkId
        }
      );
    }
  }

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "COMPLETE";

  next.validation = {
    ...(next.validation ?? {}),
    taskId,
    completedAt:
      now(),
    passed:
      true,
    checks:
      suppliedChecks
  };

  if (
    !next.tasks[taskId]
      .executionStatus
  ) {
    next.tasks[taskId]
      .executionStatus =
      "COMPLETE";
  }

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "completeTask",
      previousState,
      newState:
        "COMPLETE",
      taskScope:
        [taskId]
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "COMPLETE"
      }
    );

  return {
    transition:
      "completeTask",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    taskId,
    validationPassed:
      true,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Transition: failValidation
 * ------------------------------------------------------------
 */

function failValidation(
  state,
  rules,
  taskId,
  evidence
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "failValidation"
  );

  ensureNoMaterialPlanChange(
    state
  );

  const executingTaskId =
    state.execution?.taskId;

  const validatingTaskId =
    state.validation?.taskId;

  if (
    executingTaskId !==
      taskId ||
    validatingTaskId !==
      taskId
  ) {
    throw new StateManagerError(
      "TASK_IDENTITY_MISMATCH",
      "Validation failure task ID does not match the authoritative executing task."
    );
  }

  if (
    evidence?.validationPassed !==
    false
  ) {
    throw new StateManagerError(
      "INVALID_VALIDATION_EVIDENCE",
      "failValidation requires validationPassed=false."
    );
  }

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "AUTHORIZED";

  next.validation = {
    ...(next.validation ?? {}),
    taskId,
    completedAt:
      now(),
    passed:
      false,
    checks:
      Array.isArray(
        evidence.validationChecks
      )
        ? evidence.validationChecks
        : []
  };

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "failValidation",
      previousState,
      newState:
        "AUTHORIZED",
      taskScope:
        [taskId]
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "AUTHORIZED"
      }
    );

  return {
    transition:
      "failValidation",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    taskId,
    validationPassed:
      false,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Transition: revokeAuthorization
 * ------------------------------------------------------------
 */

function revokeAuthorization(
  state,
  rules
) {
  assertCanonicalStateConsistency(
    state
  );

  assertTransitionAllowed(
    state,
    rules,
    "revokeAuthorization"
  );

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "READY";

  next.authorization = {
    ...(next.authorization ?? {}),
    status:
      "NOT_AUTHORIZED",
    scope: [],
    revokedAt:
      now()
  };

  delete next.execution;
  delete next.validation;

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "revokeAuthorization",
      previousState,
      newState:
        "READY",
      taskScope: []
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "READY",
        authorizationStatus:
          "NOT_AUTHORIZED",
        scope: []
      }
    );

  return {
    transition:
      "revokeAuthorization",
    previousState,
    newState:
      persisted.status,
    planId:
      persisted.planId,
    planVersion:
      persisted.planVersion,
    authorizedScope:
      persisted.authorization
        .scope,
    persistedAndVerified:
      true
  };
}

/**
 * ------------------------------------------------------------
 * Controlled recovery:
 * repairAuthorizationState
 * ------------------------------------------------------------
 *
 * This operation exists only to recover a known legacy state
 * produced before the current state-transition manager became
 * authoritative.
 *
 * Allowed legacy state:
 *
 *   status = READY
 *   authorization.status = AUTHORIZED
 *   authorization.scope is non-empty
 *
 * Repair result:
 *
 *   status = READY
 *   authorization.status = NOT_AUTHORIZED
 *   authorization.scope = []
 *
 * This operation does NOT:
 *
 * - authorize a task;
 * - execute a task;
 * - modify task definitions;
 * - modify plan identity;
 * - expand scope;
 * - invoke workers.
 *
 * It is therefore a recovery operation, not a transition into
 * an execution-capable state.
 * ------------------------------------------------------------
 */

function repairAuthorizationState(
  state
) {
  ensureAuthorizationObject(
    state
  );

  const authorizationStatus =
    state.authorization.status;

  const authorizationScope =
    state.authorization.scope;

  const isRepairableLegacyState =
    state.status === "READY" &&
    authorizationStatus ===
      "AUTHORIZED" &&
    Array.isArray(
      authorizationScope
    ) &&
    authorizationScope.length >
      0;

  if (
    !isRepairableLegacyState
  ) {
    throw new StateManagerError(
      "STATE_REPAIR_NOT_APPLICABLE",
      "The canonical plan is not in the specific legacy READY + AUTHORIZED state supported by this repair operation.",
      {
        status:
          state.status,
        authorizationStatus,
        authorizationScope
      }
    );
  }

  for (
    const taskId of
      authorizationScope
  ) {
    getTask(
      state,
      taskId
    );
  }

  const next =
    deepClone(state);

  const previousState =
    next.status;

  next.status =
    "READY";

  next.authorization = {
    ...(next.authorization ?? {}),
    status:
      "NOT_AUTHORIZED",
    scope: [],
    repairedAt:
      now(),
    repairedFrom: {
      status:
        state.status,
      authorizationStatus,
      authorizationScope: [
        ...authorizationScope
      ]
    }
  };

  delete next.execution;
  delete next.validation;

  appendAudit(
    next,
    {
      actor:
        "Plan State Manager",
      action:
        "repairAuthorizationState",
      previousState,
      newState:
        "READY",
      taskScope:
        authorizationScope
    }
  );

  const persisted =
    persistAndVerify(
      next,
      {
        status:
          "READY",
        authorizationStatus:
          "NOT_AUTHORIZED",
        scope: []
      }
    );

  return {
    transition:
      "repairAuthorizationState",
    previousState,
    newState:
      persisted.status,
    previousAuthorizationStatus:
      authorizationStatus,
    newAuthorizationStatus:
      persisted.authorization
        .status,
    previousAuthorizationScope:
      authorizationScope,
    newAuthorizationScope:
      persisted.authorization
        .scope,
    persistedAndVerified:
      true,
    note:
      "Legacy authorization metadata was cleared without authorizing or executing any task."
  };
}

/**
 * ------------------------------------------------------------
 * Status
 * ------------------------------------------------------------
 */

function status(
  state,
  rules
) {
  assertKnownPlanStatus(
    state,
    rules
  );

  ensureAuthorizationObject(
    state
  );

  const legacyInconsistency =
    state.status ===
      "READY" &&
    state.authorization
      ?.status ===
      "AUTHORIZED";

  if (
    legacyInconsistency
  ) {
    return {
      command:
        "status",
      planId:
        state.planId,
      planVersion:
        state.planVersion,
      status:
        state.status,
      authorization: {
        status:
          state.authorization
            .status,
        scope: [
          ...state.authorization
            .scope
        ]
      },
      execution:
        state.execution
          ? {
              taskId:
                state.execution
                  .taskId,
              worker:
                state.execution
                  .worker,
              startedAt:
                state.execution
                  .startedAt
            }
          : null,
      validation:
        state.validation
          ? {
              taskId:
                state.validation
                  .taskId,
              passed:
                state.validation
                  .passed ??
                null,
              startedAt:
                state.validation
                  .startedAt ??
                null
            }
          : null,
      consistency:
        "LEGACY_ORPHANED_AUTHORIZATION",
      repairRequired:
        true
    };
  }

  assertCanonicalStateConsistency(
    state
  );

  return {
    command:
      "status",
    planId:
      state.planId,
    planVersion:
      state.planVersion,
    status:
      state.status,
    authorization: {
      status:
        state.authorization
          .status,
      scope: [
        ...state.authorization
          .scope
      ]
    },
    execution:
      state.execution
        ? {
            taskId:
              state.execution
                .taskId,
            worker:
              state.execution
                .worker,
            startedAt:
              state.execution
                .startedAt
          }
        : null,
    validation:
      state.validation
        ? {
            taskId:
              state.validation
                .taskId,
            passed:
              state.validation
                .passed ??
              null,
            startedAt:
              state.validation
                .startedAt ??
              null
          }
        : null,
    consistency:
      "VALID",
    repairRequired:
      false
  };
}

/**
 * ------------------------------------------------------------
 * Command-line interface
 * ------------------------------------------------------------
 */

function parseArguments(
  argv
) {
  const [
    ,
    ,
    command,
    taskId,
    evidenceJson
  ] = argv;

  if (!command) {
    throw new StateManagerError(
      "MISSING_COMMAND",
      `Usage: node ${path.relative(
        process.cwd(),
        __filename
      )} <command> [taskId] [evidenceJson]`
    );
  }

  if (
    !COMMANDS.has(command)
  ) {
    throw new StateManagerError(
      "UNKNOWN_COMMAND",
      `Unknown command '${command}'.`,
      {
        command,
        allowedCommands: [
          ...COMMANDS
        ]
      }
    );
  }

  let evidence = {};

  if (evidenceJson) {
    try {
      evidence =
        JSON.parse(
          evidenceJson
        );
    } catch (error) {
      throw new StateManagerError(
        "INVALID_EVIDENCE_JSON",
        "The supplied evidence argument is not valid JSON.",
        {
          cause:
            error.message
        }
      );
    }
  }

  return {
    command,
    taskId,
    evidence
  };
}

function main() {
  try {
    const args =
      parseArguments(
        process.argv
      );

    const rules =
      loadRules();

    const state =
      loadState();

    assertKnownPlanStatus(
      state,
      rules
    );

    let result;

    switch (
      args.command
    ) {
      case "status":
        result =
          status(
            state,
            rules
          );
        break;

      case "authorize":
        result =
          authorize(
            state,
            rules,
            args.taskId
          );
        break;

      case "beginExecution":
        result =
          beginExecution(
            state,
            rules,
            args.taskId
          );
        break;

      case "beginValidation":
        result =
          beginValidation(
            state,
            rules,
            args.taskId
          );
        break;

      case "completeTask":
        result =
          completeTask(
            state,
            rules,
            args.taskId,
            args.evidence
          );
        break;

      case "failValidation":
        result =
          failValidation(
            state,
            rules,
            args.taskId,
            args.evidence
          );
        break;

      case "revokeAuthorization":
        result =
          revokeAuthorization(
            state,
            rules
          );
        break;

      case "repairAuthorizationState":
        result =
          repairAuthorizationState(
            state
          );
        break;

      default:
        throw new StateManagerError(
          "UNKNOWN_COMMAND",
          `Unhandled command '${args.command}'.`
        );
    }

    succeed(result);
  } catch (error) {
    if (
      error instanceof
      StateManagerError
    ) {
      fail(
        error.code,
        error.message,
        error.details
      );
      return;
    }

    fail(
      "UNEXPECTED_ERROR",
      error.message ||
        String(error)
    );
  }
}

main();
